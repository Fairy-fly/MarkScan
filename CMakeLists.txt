cmake_minimum_required(VERSION 3.21)
project(MarkScan VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MARKSCAN_BUILD_UI "Build Qt application" ON)
if(ANDROID)
  find_package(OpenCV 4.6 REQUIRED COMPONENTS java)
else()
  find_package(OpenCV 4.6 REQUIRED COMPONENTS core imgproc imgcodecs)
endif()
add_library(markscan_core core/Recognizer.cpp)
target_include_directories(markscan_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(markscan_core PUBLIC ${OpenCV_LIBS})
if(MSVC)
  add_compile_options(/utf-8)
  target_compile_options(markscan_core PRIVATE /utf-8)
endif()
if(NOT ANDROID)
  add_executable(markscan_tests tests/core_tests.cpp tools/SheetFactory.cpp)
  target_link_libraries(markscan_tests PRIVATE markscan_core)
  enable_testing()
  add_test(NAME recognition COMMAND markscan_tests ${CMAKE_CURRENT_SOURCE_DIR})
endif()
if(MARKSCAN_BUILD_UI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
  qt_standard_project_setup()
  qt_add_executable(MarkScan desktop/main.cpp desktop/MainWindow.cpp desktop/Theme.cpp desktop/ImageCanvas.cpp desktop/ScoreCard.cpp android/Storage.cpp)
  target_link_libraries(MarkScan PRIVATE markscan_core Qt6::Widgets Qt6::Concurrent)
  qt_add_resources(MarkScan assets PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}"
    FILES config/answers.json templates/blank.png tests/images/all_correct.png tests/images/partial_wrong.png tests/images/unanswered.png tests/images/multiple.png tests/images/perspective.png tests/images/no_card.png)
  if(ANDROID)
    set_target_properties(MarkScan PROPERTIES QT_ANDROID_MIN_SDK_VERSION 23 QT_ANDROID_TARGET_SDK_VERSION 33
      QT_ANDROID_VERSION_CODE 2 QT_ANDROID_VERSION_NAME "1.1.0"
      QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android"
      QT_ANDROID_SDK_BUILD_TOOLS_REVISION "33.0.2"
      QT_ANDROID_EXTRA_LIBS "${OpenCV_DIR}/../libs/arm64-v8a/libopencv_java4.so"
      QT_ANDROID_PACKAGE_NAME "org.markscan.app")
  else()
    add_executable(markscan_pdf tools/print_template.cpp)
    target_link_libraries(markscan_pdf PRIVATE Qt6::Widgets)
  endif()
endif()
